---
// 呼び出し元（各ページ）から渡されるデータを受け取る
interface Props {
  current?: string; // 例: 'concept', 'staff', 'gallery', 'menu', 'news'
  isHome?: boolean; // トップページかどうか
}

const { current = '', isHome = false } = Astro.props;

// トップページなら #concept、下層ページなら /concept や /#concept に調整
const getLink = (anchor: string, path: string) => {
  return isHome ? `#${anchor}` : path;
};
